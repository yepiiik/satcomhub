29static libsgp4::DateTime toSgp4DateTime(const QDateTime &qdt) {
30 QDateTime utc = qdt.isValid() ? qdt.toUTC() : QDateTime::currentDateTimeUtc();
31 QDate d = utc.date();
32 QTime t = utc.time();
33 int y = d.year() > 0 ? d.year() : 2026;
34 int m = (d.month() >= 1 && d.month() <= 12) ? d.month() : 1;
35 int day = (d.day() >= 1 && d.day() <= 31) ? d.day() : 1;
36 return libsgp4::DateTime(y, m, day,
37 t.hour(), t.minute(), t.second(),
38 t.msec() * 1000);
39}

97bool SatellitePropagator::extractTleLines(const QString &inLine1, const QString &inLine2,
98 QString &outName, QString &outLine1, QString &outLine2) {
99 QString l1 = inLine1.trimmed();
100 QString l2 = inLine2.trimmed();
101 outName.clear();
102
103 // Support multi-line or 3-line TLE blocks pasted into inLine1
104 if (l1.contains('\n') || l1.contains('\r')) {
105 QStringList lines = l1.split(QRegularExpression("[\r\n]+"), Qt::SkipEmptyParts);
106 for (const QString &rawLine : lines) {
107 QString trimmed = rawLine.trimmed();
108 if (trimmed.startsWith('1') && trimmed.length() >= 60) {
109 l1 = trimmed;
110 } else if (trimmed.startsWith('2') && trimmed.length() >= 60) {
111 l2 = trimmed;
112 } else if (!trimmed.isEmpty() && outName.isEmpty()) {
113 outName = trimmed;
114 }
115 }
116 }
117
118 if (!l1.startsWith('1') || !l2.startsWith('2') || l1.length() < 60 || l2.length() < 60) {
119 return false;
120 }
121
122 // Standard TLE line length expected by libsgp4 is strictly 69 characters.
123 // If copy-paste stripped trailing whitespace, pad to 69 characters.
124 if (l1.length() < 69) {
125 l1 = l1.leftJustified(69, ' ');
126 } else if (l1.length() > 69) {
127 l1 = l1.left(69);
128 }
129
130 if (l2.length() < 69) {
131 l2 = l2.leftJustified(69, ' ');
132 } else if (l2.length() > 69) {
133 l2 = l2.left(69);
134 }
135
136 outLine1 = l1;
137 outLine2 = l2;
138 if (outName.isEmpty()) {
139 outName = QStringLiteral("SATELLITE");
140 }
141 return true;
142}

144double SatellitePropagator::calculateVisibilityRadius(double altitudeKm, double minElevationDeg) const {
145 if (altitudeKm <= 0.0) return 0.0;
146
147 double elRad = minElevationDeg * (M_PI / 180.0);
148 double rSat = EarthRadiusKm + altitudeKm;
149
150 // Earth central angle theta from spherical trigonometry
151 double cosArg = (EarthRadiusKm / rSat) * std::cos(elRad);
152 if (cosArg > 1.0) cosArg = 1.0;
153 if (cosArg < -1.0) cosArg = -1.0;
154
155 double theta = std::acos(cosArg) - elRad;
156 if (theta < 0.0) theta = 0.0;
157
158 // Arc distance along Earth surface in meters
159 return (EarthRadiusKm * 1000.0) * theta;
160}

207 double startMinutesFromNow, double endMinutesFromNow, double stepMinutes) const {
208 QVariantList track;
209 double step = std::max(0.1, stepMinutes);
210 for (double m = startMinutesFromNow; m <= endMinutesFromNow; m += step) {
211 QDateTime stepUtc = baseUtc.addMSecs(static_cast<qint64>(m * 60000.0));
212 libsgp4::DateTime dt = toSgp4DateTime(stepUtc);
213
214 try {
215 libsgp4::Eci eci = sgp4.FindPosition(dt);

218 QVariantMap pt;
219 pt["latitude"] = libsgp4::Util::RadiansToDegrees(geo.latitude);
220 pt["longitude"] = libsgp4::Util::RadiansToDegrees(geo.longitude);
221 pt["altitudeKm"] = geo.altitude;
222 pt["minuteOffset"] = m;

256
257 double meanMotion = tle.MeanMotion(); // rev / day
258 if (meanMotion <= 0.0) return res;
259

261 double half = periodMinutes / 2.0;
262 QDateTime nowUtc = QDateTime::currentDateTimeUtc();
263
264 res["periodMinutes"] = periodMinutes;
265 res["previous"] = computeTrackRange(sgp4, nowUtc, -half - periodMinutes, -half, stepMinutes);
266 res["current"] = computeTrackRange(sgp4, nowUtc, -half, half, stepMinutes);
267 res["next"] = computeTrackRange(sgp4, nowUtc, half, half + periodMinutes, stepMinutes);
268 return res;
269 } catch (...) {

337 // Bisection helper to find exact time when elevation crosses threshold
338 auto refineCrossing = [&](libsgp4::Observer &obs, const QDateTime &tBelow, const QDateTime &tAbove, double thresholdRad) -> QDateTime {
339 QDateTime low = tBelow;
340 QDateTime high = tAbove;
341 for (int iter = 0; iter < 5; ++iter) {
342 qint64 midMs = (low.toMSecsSinceEpoch() + high.toMSecsSinceEpoch()) / 2;
343 QDateTime mid = QDateTime::fromMSecsSinceEpoch(midMs, QTimeZone::UTC);
344 libsgp4::DateTime dtMid = toSgp4DateTime(mid);
345 libsgp4::Eci eciMid = sgp4.FindPosition(dtMid);
346 libsgp4::CoordTopocentric topoMid = obs.GetLookAngle(eciMid);
347 if (topoMid.elevation >= thresholdRad) {
348 high = mid;
349 } else {
350 low = mid;
351 }
352 }
353 return high;

356 // 1. Calculate individual passes for each station
357 for (auto &st : stationObservers) {

359 QDateTime aosTime;
360 QDateTime tcaTime;

362 double aosAzRad = 0.0;
363 double tcaAzRad = 0.0;

371 libsgp4::DateTime dt = toSgp4DateTime(currTime);

373 double elRad = 0.0;
374 double azRad = 0.0;
375 double rangeKm = 0.0;
376 try {
377 libsgp4::Eci eci = sgp4.FindPosition(dt);
378 libsgp4::CoordTopocentric topo = st.observer->GetLookAngle(eci);
379 elRad = topo.elevation;
380 azRad = topo.azimuth;
381 rangeKm = topo.range;
382 } catch (...) {

384 }

386 if (elRad >= st.minElRad) {

389 // Refine AOS

392 } else {

406 } else {
407 if (inPass) {
408 inPass = false;
409 // Refine LOS
410 QDateTime losTime;
411 if (prevElRad >= st.minElRad) {
412 losTime = refineCrossing(*st.observer, currTime, prevTime, st.minElRad);
413 } else {
414 losTime = currTime;
415 }
416
417 qint64 durSec = aosTime.secsTo(losTime);
418 if (durSec >= 30) {

420 p.stationId = st.stationId;
421 p.stationName = st.name;
422 p.aosUtc = aosTime;
423 p.losUtc = losTime;
424 p.tcaUtc = tcaTime;
425 p.maxElDeg = libsgp4::Util::RadiansToDegrees(maxElRad);

438 prevElRad = elRad;
439 }

447 // 3. Group overlapping / handover passes from different stations into Joint Pass Events
448 struct PassGroup {

454 std::vector<PassGroup> passGroups;

523 eventMap["maxElevationStation"] = maxElStationName;

533 eventMap["windowGainPercent"] = windowGainPercent;
534
535 // Check for concurrent overlap (stations seeing satellite simultaneously for High-SNR MRC)
536 bool hasConcurrentOverlap = false;
537 qint64 maxOverlapSec = 0;
538 if (isMultiStation && grp.passes.size() >= 2) {

541 QDateTime oStart = std::max(grp.passes[i].aosUtc, grp.passes[j].aosUtc);
542 QDateTime oEnd = std::min(grp.passes[i].losUtc, grp.passes[j].losUtc);
543 if (oStart < oEnd) {
544 hasConcurrentOverlap = true;
545 qint64 oDur = oStart.secsTo(oEnd);
546 if (oDur > maxOverlapSec) {
547 maxOverlapSec = oDur;
548 }
549 }
550 }
551 }
552 }
553 eventMap["hasConcurrentOverlap"] = hasConcurrentOverlap;
554 eventMap["overlapDurationSeconds"] = maxOverlapSec;
555 eventMap["overlapDurationFormatted"] = formatDuration(maxOverlapSec);
556
557 // Sub-passes breakdown
558 QVariantList stationPassesList;
559 for (const auto &sp : grp.passes) {
560 QVariantMap spMap;
561 spMap["stationId"] = sp.stationId;
562 spMap["stationName"] = sp.stationName;
563 spMap["aosUtc"] = sp.aosUtc.toString("HH:mm:ss");
564 spMap["losUtc"] = sp.losUtc.toString("HH:mm:ss");
565 spMap["tcaUtc"] = sp.tcaUtc.toString("HH:mm:ss");

574 }
575 eventMap["stationPasses"] = stationPassesList;
576
577 formattedPasses.append(eventMap);
578 }

581 result["passes"] = formattedPasses;
582 result["totalPasses"] = formattedPasses.size();
583 result["multiStationPasses"] = multiStationCount;

628double SatellitePropagator::calculateRMax(double snrDb, double bitsPerSymbol, double alpha, double codeEfficiencyCoef) const {
629 if (bitsPerSymbol <= 0.0 || (1.0 + alpha) <= 0.0 || codeEfficiencyCoef <= 0.0) {
630 return 0.0;
631 }
632 if (snrDb <= -25.0) {
633 return 0.0;
634 }
635
636 double snrLinear = std::pow(10.0, snrDb / 10.0);
637 double arg = 1.0 + (snrLinear / codeEfficiencyCoef);
638 if (arg <= 1.0) {
639 return 0.0;
640 }
641
642 double prefactor = 1.0 / (bitsPerSymbol * (1.0 + alpha));
643 double rMax = prefactor * std::log2(arg);
644 return std::max(0.0, rMax);
645}

647QString SatellitePropagator::generateSessionPlanYaml(const QVariantMap &sessionEvent,
648 const QVariantMap &snrProfileData,
649 const QVariantMap &transmissionParams) const {
650 QString yaml;
651 QTextStream out(&yaml);
652
653 // 1. Static Configuration Parameters
654 double carrierFreqMhz = transmissionParams.value("carrierFreqMhz",
655 snrProfileData.value("carrierFreqMhz", 5840.0).toDouble()).toDouble();
656 if (carrierFreqMhz <= 0.0) carrierFreqMhz = 5840.0;
657
658 double channelBandwidthKhz = transmissionParams.value("channelBandwidthKhz",
659 snrProfileData.value("channelBandwidthKhz", 10000.0).toDouble()).toDouble();
660 if (channelBandwidthKhz <= 0.0) channelBandwidthKhz = 10000.0;
661
662 double bwMhz = channelBandwidthKhz / 1000.0;
663 QString cFreqStr = (carrierFreqMhz == std::floor(carrierFreqMhz))
664 ? QString::number(static_cast<qint64>(carrierFreqMhz))
665 : QString::number(carrierFreqMhz, 'g', 10);
666 QString bwStr = (bwMhz == std::floor(bwMhz))
667 ? QString::number(static_cast<qint64>(bwMhz))
668 : QString::number(bwMhz, 'g', 10);
669
670 out << "static_config: |-\n";
671 out << " --c_freq=" << cFreqStr << "\n";
672 out << " --bandwidth=" << bwStr << "\n";
673
674 // 2. Transmission & AMC Parameters
675 double bitsPerSymbol = transmissionParams.value("bitsPerSymbol", 2.0).toDouble();
676 if (bitsPerSymbol <= 0.0) bitsPerSymbol = 2.0;
677
678 double alpha = transmissionParams.value("alpha", 0.20).toDouble();
679 double codeEfficiencyCoef = transmissionParams.value("codeEfficiencyCoef", 1.00).toDouble();
680 if (codeEfficiencyCoef <= 0.0) codeEfficiencyCoef = 1.0;
681
682 QString registerPath = transmissionParams.value("codeRateRegister", "fpgatrx/code_rate").toString().trimmed();
683 if (registerPath.isEmpty()) registerPath = "fpgatrx/code_rate";
684
685 // 3. Hardware Code Rate Tiers
686 struct Tier {
687 QString name;
688 double rate = 0.5;
689 QString hw = "0";
690 };
691 std::vector<Tier> tiers;
692
693 QVariantList tableList = transmissionParams.value("codeRateTable").toList();
694 if (!tableList.isEmpty()) {
695 for (const QVariant &itemVar : tableList) {
696 QVariantMap item = itemVar.toMap();
697 Tier t;
698 t.name = item.value("name", "").toString();
699 t.rate = item.value("rate", 0.5).toDouble();
700 t.hw = item.value("hw", "0").toString();
701 if (t.rate > 0.0) {
702 tiers.push_back(t);
703 }
704 }
705 }
706
707 // Default to Polyitan-3 mission tiers if none provided
708 if (tiers.empty()) {
709 tiers = {
710 {"R1", 0.910, "0"},
711 {"R2", 0.830, "1"},
712 {"R3", 0.700, "2"},
713 {"R4", 0.570, "3"},
714 {"R5", 0.410, "4"},
715 {"R6", 0.285, "5"},
716 {"R7", 0.190, "6"}
717 };
718 }
719
720 // Sort descending by rate: R1 (0.91) down to R7 (0.19)
721 std::sort(tiers.begin(), tiers.end(), [](const Tier &a, const Tier &b) {
722 return a.rate > b.rate;
723 });
724
725 auto selectOptimalTier = [&](double rMaxVal) -> Tier {
726 for (const auto &t : tiers) {
727 if (t.rate <= rMaxVal) {
728 return t;
729 }
730 }
731 // If rMaxVal is lower than all tiers, select the most robust tier (last in sorted descending)
732 return tiers.back();
733 };
734
735 // 4. Time Base and Epoch Extraction
736 QDateTime startBaseUtc;
737 QString startIso = sessionEvent.value("combinedAosIso").toString();
738 if (!startIso.isEmpty()) {
739 startBaseUtc = QDateTime::fromString(startIso, Qt::ISODate);
740 }
741 if (!startBaseUtc.isValid()) {
742 QString startUtcStr = sessionEvent.value("combinedAosUtc", snrProfileData.value("startUtc", "")).toString();
743 startBaseUtc = QDateTime::fromString(startUtcStr, "yyyy-MM-dd HH:mm:ss 'UTC'");
744 if (!startBaseUtc.isValid()) {
745 startBaseUtc = QDateTime::fromString(startUtcStr, "yyyy-MM-dd HH:mm:ss");
746 }
747 }
748 if (!startBaseUtc.isValid()) {
749 startBaseUtc = QDateTime::currentDateTimeUtc();
750 }
751 startBaseUtc.setTimeZone(QTimeZone::UTC);
752
753 QVariantList selSnrList = snrProfileData.value("combinedSelectionSnrDb").toList();
754 QVariantList timestampsList = snrProfileData.value("timestampsSec").toList();
755
756 out << "session_plan:\n";
757
758 if (timestampsList.isEmpty() || selSnrList.isEmpty()) {
759 // Fallback single entry at start
760 Tier fallbackTier = tiers.back();
761 out << " - epoch: " << startBaseUtc.toString("yyyy-MM-dd HH:mm:ss") << "\n";
762 out << " config:\n";
763 out << " - " << fallbackTier.hw << " > " << registerPath << "\n";
764 return yaml;
765 }
766
767 // 5. Evaluate optimal code rate across the pass timeline and record transition epochs
768 int sampleCount = std::min(timestampsList.size(), selSnrList.size());
769 QString currentHw = "";
770 int lastTransitionSample = -999;
771 constexpr int minDwellSamples = 10; // At 1 Hz sample rate, 10s dwell prevents ping-pong near boundaries
772
773 for (int i = 0; i < sampleCount; ++i) {
774 double relSec = timestampsList[i].toDouble();
775 double snr = selSnrList[i].toDouble();
776
777 double rMax = (snr > -25.0) ? calculateRMax(snr, bitsPerSymbol, alpha, codeEfficiencyCoef) : 0.0;
778 Tier optTier = selectOptimalTier(rMax);
779
780 bool isFirst = (i == 0);
781 bool tierChanged = (optTier.hw != currentHw);
782
783 if (isFirst || (tierChanged && (i - lastTransitionSample >= minDwellSamples))) {
784 currentHw = optTier.hw;
785 lastTransitionSample = i;
786
787 QDateTime epochTime = startBaseUtc.addSecs(static_cast<qint64>(std::round(relSec)));
788 out << " - epoch: " << epochTime.toString("yyyy-MM-dd HH:mm:ss") << "\n";
789 out << " config:\n";
790 out << " - " << currentHw << " > " << registerPath << "\n";
791 }
792 }
793
794 return yaml;
795}

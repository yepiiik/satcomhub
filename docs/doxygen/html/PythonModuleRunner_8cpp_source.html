36static libsgp4::DateTime toSgp4DateTime(const QDateTime &qdt) {
37 QDateTime utc = qdt.isValid() ? qdt.toUTC() : QDateTime::currentDateTimeUtc();
38 QDate d = utc.date();
39 QTime t = utc.time();
40 int y = d.year() > 0 ? d.year() : 2026;
41 int m = (d.month() >= 1 && d.month() <= 12) ? d.month() : 1;
42 int day = (d.day() >= 1 && d.day() <= 31) ? d.day() : 1;
43 return libsgp4::DateTime(y, m, day, t.hour(), t.minute(), t.second(), t.msec() * 1000);
44}

46static bool extractTle(const QString &in1, const QString &in2,
47 QString &name, QString &line1, QString &line2) {
48 QString t1 = in1.trimmed();
49 QString t2 = in2.trimmed();
50
51 if (t1.startsWith("1 ") && t2.startsWith("2 ")) {
52 name = "SAT";
53 line1 = t1;
54 line2 = t2;
55 return true;
56 }
57
58 QStringList lines = (t1 + "\n" + t2).split(QRegularExpression("[\r\n]+"), Qt::SkipEmptyParts);
59 for (int i = 0; i < lines.size(); ++i) {
60 QString trimmed = lines[i].trimmed();
61 if (trimmed.startsWith("1 ") && (i + 1 < lines.size()) && lines[i + 1].trimmed().startsWith("2 ")) {
62 name = (i > 0) ? lines[i - 1].trimmed() : "SAT";
63 line1 = trimmed;
64 line2 = lines[i + 1].trimmed();
65 return true;
66 }
67 }
68 return false;
69}

155QVariantMap PythonModuleRunner::detectPythonEnvironment() const
156{
157 QVariantMap info;
158 bool isVenv = false;
159 QString exe = resolvePythonExecutable(&isVenv);
160
161 if (exe.isEmpty()) {
162 info["available"] = false;
163 info["executable"] = "";
164 info["version"] = "Not Found";
165 info["isVenv"] = false;
166 return info;
167 }
168
169 QProcess proc;
170 proc.start(exe, {"--version"});
171 proc.waitForFinished(1500);
172 QString verOut = QString::fromUtf8(proc.readAllStandardOutput()).trimmed();
173 if (verOut.isEmpty()) {
174 verOut = QString::fromUtf8(proc.readAllStandardError()).trimmed();
175 }
176
177 info["available"] = (proc.exitStatus() == QProcess::NormalExit && proc.exitCode() == 0);
178 info["executable"] = exe;
179 info["version"] = verOut.isEmpty() ? "Python 3" : verOut;
180 info["isVenv"] = isVenv;
181 return info;

631QString PythonModuleRunner::buildSessionInputJson(
632 const QString &tleLine1,
633 const QString &tleLine2,
634 const QVariantList &stations,
635 const QString &startUtcIso,
636 const QString &endUtcIso,
637 double sampleRateHz)
638{
639 QDateTime startUtc = QDateTime::fromString(startUtcIso, Qt::ISODate);
640 if (!startUtc.isValid()) {
641 startUtc = QDateTime::fromString(startUtcIso, "yyyy-MM-dd HH:mm:ss 'UTC'");
642 if (!startUtc.isValid()) {
643 startUtc = QDateTime::fromString(startUtcIso, "yyyy-MM-dd HH:mm:ss");
644 }
645 }
646 startUtc.setTimeZone(QTimeZone::UTC);
647
648 QDateTime endUtc = QDateTime::fromString(endUtcIso, Qt::ISODate);
649 if (!endUtc.isValid()) {
650 endUtc = QDateTime::fromString(endUtcIso, "yyyy-MM-dd HH:mm:ss 'UTC'");
651 if (!endUtc.isValid()) {
652 endUtc = QDateTime::fromString(endUtcIso, "yyyy-MM-dd HH:mm:ss");
653 }
654 }
655 endUtc.setTimeZone(QTimeZone::UTC);
656
657 QJsonArray satSessionArray;
658
659 QString satName, l1, l2;
660 if (extractTle(tleLine1, tleLine2, satName, l1, l2) && startUtc.isValid() && endUtc.isValid() && startUtc < endUtc) {
661 try {
662 libsgp4::Tle tle(satName.toStdString(), l1.toStdString(), l2.toStdString());
663 libsgp4::SGP4 sgp4(tle);
664
665 double safeRate = (sampleRateHz <= 0.0) ? 1.0 : std::clamp(sampleRateHz, 0.1, 10.0);
666 double stepSec = 1.0 / safeRate;
667 qint64 stepMs = static_cast<qint64>(std::round(stepSec * 1000.0));
668 if (stepMs < 100) stepMs = 1000;
669
670 qint64 totalMs = startUtc.msecsTo(endUtc);
671 int numSteps = static_cast<int>(totalMs / stepMs) + 1;
672 if (numSteps > 3600) numSteps = 3600;
673
674 for (int i = 0; i < numSteps; ++i) {
675 QDateTime sampleDt = startUtc.addMSecs(static_cast<qint64>(i) * stepMs);
676 libsgp4::DateTime sgpDt = toSgp4DateTime(sampleDt);
677 libsgp4::Eci eci = sgp4.FindPosition(sgpDt);
678 libsgp4::CoordGeodetic geo = eci.ToGeodetic();
679
680 double latDeg = libsgp4::Util::RadiansToDegrees(geo.latitude);
681 double lonDeg = libsgp4::Util::RadiansToDegrees(geo.longitude);
682 double altKm = geo.altitude;
683
684 QJsonObject ptObj;
685 ptObj["utc_time"] = sampleDt.toString("dd-MM-yyyy HH:mm:ss");
686 ptObj["longitude"] = std::round(lonDeg * 100000.0) / 100000.0;
687 ptObj["latitiude"] = std::round(latDeg * 100000.0) / 100000.0;
688 ptObj["altitude"] = std::round(altKm * 10.0) / 10.0;
689 satSessionArray.append(ptObj);
690 }
691 } catch (...) {
692 }
693 }
694
695 StationManager stMgr;
696 QJsonArray stationsArray;
697 for (const QVariant &stVar : stations) {
698 QVariantMap stMap = stVar.toMap();
699 QString stJsonStr = stMgr.buildStationJson(stMap);
700 QJsonDocument stDoc = QJsonDocument::fromJson(stJsonStr.toUtf8());
701 if (stDoc.isObject()) {
702 stationsArray.append(stDoc.object());
703 }
704 }
705
706 QJsonObject root;
707 root["satellite_session"] = satSessionArray;
708 root["ground_stations_list"] = stationsArray;
709
710 return QString::fromUtf8(QJsonDocument(root).toJson(QJsonDocument::Indented));
711}

719 QJsonArray factorsList;
720 int determinedSamples = expectedSampleCount > 0 ? expectedSampleCount : 0;
721
722 // First pass: parse factor module outputs and determine sample count for this station
723 struct ParsedFactor {
724 QString factorName;
725 QString factorUnits;
726 QJsonArray values;
727 };
728 QList<ParsedFactor> parsedFactors;
729
730 for (const QVariant &v : factorModuleOutputs) {
731 QString jsonStr;
732 if (v.typeId() == QMetaType::QString) {
733 jsonStr = v.toString();
734 } else if (v.canConvert<QVariantMap>()) {
735 jsonStr = v.toMap().value("outputJson").toString();
736 }
737 if (jsonStr.trimmed().isEmpty()) continue;
738
739 QJsonParseError err;
740 QJsonDocument doc = QJsonDocument::fromJson(jsonStr.toUtf8(), &err);
741 if (err.error != QJsonParseError::NoError || !doc.isObject()) continue;
742
743 QJsonObject obj = doc.object();
744 QString factorName = obj.value("factor_name").toString().trimmed();
745 if (factorName.isEmpty()) continue;
746
747 QString factorUnits = obj.value("units").toString().trimmed();
748 if (factorUnits.isEmpty()) {
749 factorUnits = obj.value("factor_units").toString().trimmed();
750 }
751
752 QJsonArray vals;
753 if (obj.contains("ground_stations_session_values") && obj.value("ground_stations_session_values").isArray()) {
754 QJsonArray stArr = obj.value("ground_stations_session_values").toArray();
755 QJsonObject matchedStationObj;
756 bool foundMatch = false;
757
758 // 1. Match by stationId and/or ground_station_name
759 for (const QJsonValue &stVal : stArr) {
760 if (!stVal.isObject()) continue;
761 QJsonObject stObj = stVal.toObject();
762 QString candId = stObj.value("stationId").toString().trimmed();
763 QString candName = stObj.value("ground_station_name").toString().trimmed();
764 if ((!stationId.isEmpty() && candId == stationId) ||
765 (!groundStationName.isEmpty() && candName.compare(groundStationName, Qt::CaseInsensitive) == 0)) {
766 matchedStationObj = stObj;
767 foundMatch = true;
768 break;
769 }
770 }
771 // Fallback to first entry if only one station is present in output
772 if (!foundMatch && !stArr.isEmpty() && stArr.first().isObject()) {
773 matchedStationObj = stArr.first().toObject();
774 foundMatch = true;
775 }
776
777 if (foundMatch) {
778 QJsonArray sessArr = matchedStationObj.value("session_values").toArray();
779 for (const QJsonValue &sv : sessArr) {
780 if (sv.isObject()) {
781 vals.append(sv.toObject().value("value"));
782 } else {
783 vals.append(sv);
784 }
785 }
786 }
787 } else if (obj.contains("session_values") && obj.value("session_values").isArray()) {
788 QJsonArray sessArr = obj.value("session_values").toArray();
789 for (const QJsonValue &sv : sessArr) {
790 if (sv.isObject()) {
791 vals.append(sv.toObject().value("value"));
792 } else {
793 vals.append(sv);
794 }
795 }
796 } else if (obj.contains("values") && obj.value("values").isArray()) {
797 vals = obj.value("values").toArray();
798 }
799
800 if (!vals.isEmpty()) {
801 determinedSamples = vals.size();
802 }
803 parsedFactors.append({factorName, factorUnits, vals});
804 }
805
806 for (const ParsedFactor &pf : parsedFactors) {
807 QJsonArray normalizedVals;
808 for (int i = 0; i < determinedSamples; ++i) {
809 if (i < pf.values.size()) {
810 normalizedVals.append(pf.values.at(i));
811 } else {
812 normalizedVals.append(0.0);
813 }
814 }
815
816 QJsonObject factorObj;
817 factorObj["factor_name"] = pf.factorName;
818 factorObj["factor_units"] = pf.factorUnits;
819 factorObj["values"] = normalizedVals;
820 factorsList.append(factorObj);
821 }
822
823 QJsonObject root;
824 root["ground_station_name"] = groundStationName;
825 root["samples_count"] = determinedSamples;
826 root["factors_list"] = factorsList;
827
828 return QString::fromUtf8(QJsonDocument(root).toJson(QJsonDocument::Indented));
829}

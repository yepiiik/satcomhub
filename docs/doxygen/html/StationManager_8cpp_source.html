76QVariantMap StationManager::computeDerivedParameters(
77 const QString &band,
78 double lnaNfDb,
79 double feederLossDb,
80 double antennaGainDbi,
81 double customFMinMhz,
82 double customFMaxMhz,
83 double antennaSkyNoiseK)
84{
85 QVariantMap res;
86
87 // 1. Operating Band Frequencies & Channel Width
88 double fMinMhz = 0.0;
89 double fMaxMhz = 0.0;
90 getBandFrequencies(band, customFMinMhz, customFMaxMhz, fMinMhz, fMaxMhz);
91 double maxChannelWidthMhz = std::max(0.0, fMaxMhz - fMinMhz);
92
93 // 2. Average Frequency & Wavelength
94 double fAvgMhz = (fMinMhz + fMaxMhz) / 2.0;
95 if (fAvgMhz <= 0.0) fAvgMhz = 4000.0;
96 const double c = 299792458.0; // speed of light in m/s
97 double wavelength = c / (fAvgMhz * 1e6); // in meters
98
99 // 3. Dish Diameter (D) [m] from antenna gain and avg frequency (aperture efficiency eta = 0.55)
100 // G_lin = eta * (pi * D / lambda)^2 => D = (lambda / pi) * sqrt(G_lin / eta)
101 const double eta = 0.55;
102 double gLinear = std::pow(10.0, antennaGainDbi / 10.0);
103 double dishDiameterM = (wavelength / M_PI) * std::sqrt(gLinear / eta);
104 if (dishDiameterM < 0.01) dishDiameterM = 0.01;
105
106 // 4. Half-Power Beamwidth HPBW [deg]: theta ~ 70 * (lambda / D)
107 double beamwidthHpbwDeg = 70.0 * (wavelength / dishDiameterM);
108
109 // 5. Friis Noise Calculations:
110 // Total Cascade Noise Figure = LNA NF + Feeder Loss (feeder before LNA)
111 double safeLnaNf = std::max(0.0, lnaNfDb);
112 double safeLoss = std::max(0.0, feederLossDb);
113 double totalNoiseFigureDb = safeLnaNf + safeLoss;
114
115 // Linear noise factor F = 10^(NF / 10)
116 double fLinear = std::pow(10.0, totalNoiseFigureDb / 10.0);
117
118 // Receiver noise temperature Te = T0 * (F - 1), where T0 = 290 K standard reference
119 const double T0 = 290.0;
120 double receiverNoiseTempK = T0 * (fLinear - 1.0);
121 if (receiverNoiseTempK < 0.0) receiverNoiseTempK = 0.0;
122
123 // Clear sky background antenna noise temperature
124 double TantK = (antennaSkyNoiseK > 0.0) ? antennaSkyNoiseK : 40.0;
125 double systemNoiseTempK = receiverNoiseTempK + TantK;
126
127 // 6. Antenna G/T Figure of Merit [dB/K] = G [dBi] - 10 * log10(Tsys)
128 double gOverT = 0.0;
129 if (systemNoiseTempK > 0.0) {
130 gOverT = antennaGainDbi - 10.0 * std::log10(systemNoiseTempK);
131 }
132
133 res["operating_band"] = band;
134 res["freq_min_mhz"] = std::round(fMinMhz * 100.0) / 100.0;
135 res["freq_max_mhz"] = std::round(fMaxMhz * 100.0) / 100.0;
136 res["max_channel_width_mhz"] = std::round(maxChannelWidthMhz * 100.0) / 100.0;
137
138 res["dish_diameter_m"] = std::round(dishDiameterM * 100.0) / 100.0;
139 res["beamwidth_hpbw_deg"] = std::round(beamwidthHpbwDeg * 100.0) / 100.0;
140
141 res["lna_noise_figure_db"] = std::round(safeLnaNf * 100.0) / 100.0;
142 res["feeder_loss_db"] = std::round(safeLoss * 100.0) / 100.0;
143 res["total_noise_figure_db"] = std::round(totalNoiseFigureDb * 100.0) / 100.0;
144 res["receiver_noise_temp_k"] = std::round(receiverNoiseTempK * 10.0) / 10.0;
145 res["antenna_noise_temp_k"] = TantK;
146 res["system_noise_temp_k"] = std::round(systemNoiseTempK * 10.0) / 10.0;
147
148 res["antenna_gain_dbi"] = std::round(antennaGainDbi * 10.0) / 10.0;
149 res["g_over_t_db_k"] = std::round(gOverT * 100.0) / 100.0;
150
151 return res;
152}

870QVariantMap StationManager::parseSatelliteJson(const QString &jsonString)
871{
872 QVariantMap result;
873 result["valid"] = false;
874
875 QJsonParseError err;
876 QJsonDocument doc = QJsonDocument::fromJson(jsonString.trimmed().toUtf8(), &err);
877 if (err.error != QJsonParseError::NoError || !doc.isObject()) {
878 return result;
879 }
880
881 QJsonObject root = doc.object();
882 result["valid"] = true;
883 result["satId"] = root.value("satId").toString();
884 result["name"] = root.value("name").toString();
885
886 if (root.contains("tle") && root["tle"].isObject()) {
887 QJsonObject tle = root["tle"].toObject();
888 result["tle1"] = tle.value("line1").toString();
889 result["tle2"] = tle.value("line2").toString();
890 } else {
891 result["tle1"] = root.value("tle1").toString();
892 result["tle2"] = root.value("tle2").toString();
893 }
894
895 if (root.contains("rf_configuration") && root["rf_configuration"].isObject()) {
896 QJsonObject rf = root["rf_configuration"].toObject();
897 result["carrierFreqMhz"] = rf.value("carrier_frequency_mhz").toDouble(5850.0);
898 result["channelBandwidthKhz"] = rf.value("channel_bandwidth_khz").toDouble(20000.0);
899 result["modulation"] = rf.value("modulation").toString("OQPSK");
900 result["bitsPerSymbol"] = rf.value("bits_per_symbol").toDouble(2.0);
901 result["codeEfficiencyCoef"] = rf.value("code_efficiency_coef").toDouble(1.24);
902 result["rolloffAlpha"] = rf.value("rolloff_alpha").toDouble(0.35);
903 result["txPowerDbm"] = rf.value("tx_power_dbm").toDouble(33.0);
904 result["txAntennaGainDbi"] = rf.value("tx_antenna_gain_dbi").toDouble(3.0);
905 } else {
906 result["carrierFreqMhz"] = root.value("carrierFreqMhz").toDouble(5850.0);
907 result["channelBandwidthKhz"] = root.value("channelBandwidthKhz").toDouble(20000.0);
908 result["modulation"] = root.value("modulation").toString("OQPSK");
909 result["bitsPerSymbol"] = root.value("bitsPerSymbol").toDouble(2.0);
910 result["codeEfficiencyCoef"] = root.value("codeEfficiencyCoef").toDouble(1.24);

912 result["txPowerDbm"] = root.value("txPowerDbm").toDouble(33.0);
913 result["txAntennaGainDbi"] = root.value("txAntennaGainDbi").toDouble(3.0);
914 }
915
916 if (root.contains("coding") && root["coding"].isObject()) {
917 QJsonObject cd = root["coding"].toObject();
918 result["codeRateRegister"] = cd.value("code_rate_register").toString();
919 if (cd.contains("code_rate_table") && cd["code_rate_table"].isArray()) {
920 QVariantList crList;
921 QJsonArray crArr = cd["code_rate_table"].toArray();

933 }
934 } else {
935 result["codeRateRegister"] = root.value("codeRateRegister").toString();
936 result["codeRateTable"] = root.value("codeRateTable").toVariant().toList();
937 }
938
939 result["satelliteJson"] = buildSatelliteJson(result);
940 return result;

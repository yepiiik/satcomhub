30static bool extractTle(const QString &in1, const QString &in2,
31 QString &name, QString &line1, QString &line2) {
32 QString t1 = in1.trimmed();
33 QString t2 = in2.trimmed();
34
35 if (t1.startsWith("1 ") && t2.startsWith("2 ")) {
36 name = "SAT";
37 line1 = t1;
38 line2 = t2;
39 return true;
40 }
41
42 QStringList lines = (t1 + "\n" + t2).split(QRegularExpression("[\r\n]+"), Qt::SkipEmptyParts);
43 for (int i = 0; i < lines.size(); ++i) {
44 QString trimmed = lines[i].trimmed();
45 if (trimmed.startsWith("1 ") && (i + 1 < lines.size()) && lines[i + 1].trimmed().startsWith("2 ")) {
46 name = (i > 0) ? lines[i - 1].trimmed() : "SAT";
47 line1 = trimmed;
48 line2 = lines[i + 1].trimmed();
49 return true;
50 }
51 }
52 return false;
53}

56double SignalLossModel::calculateFspl(double slantRangeKm, double freqMhz) {
57 if (slantRangeKm <= 0.0 || freqMhz <= 0.0) return 0.0;
58 // Standard FSPL formula: 20*log10(d_km) + 20*log10(f_MHz) + 32.4418
59 return 20.0 * std::log10(slantRangeKm) + 20.0 * std::log10(freqMhz) + 32.4418;
60}

62double SignalLossModel::calculateAtmosphericLoss(double elevationDeg, double freqMhz) {
63 Q_UNUSED(freqMhz);
64 if (elevationDeg <= 0.0) return 3.0; // High attenuation near 0 horizon
65 // Tropospheric cosecant law with zenith attenuation of ~0.075 dB at C-band
66 double safeElRad = std::max(4.0, elevationDeg) * (M_PI / 180.0);
67 double loss = 0.075 / std::sin(safeElRad);
68 return std::clamp(loss, 0.05, 3.5);
69}

71double SignalLossModel::calculateLinkBudgetSnr(double txPowerDbm, double txAntennaGainDbi,
72 double totalLossDb, double gOverTDbK,
73 double channelBandwidthKhz) {
74 if (channelBandwidthKhz <= 0.0) return -99.0;
75
76 // 1. Satellite EIRP [dBW] = (P_tx [dBm] - 30) + G_tx [dBi]
77 double eirpDbw = (txPowerDbm - 30.0) + txAntennaGainDbi;
78
79 // 2. Boltzmann constant: k_B = 1.380649e-23 J/K -> -10*log10(k_B) = +228.599 dBW/(Hz*K)
80 constexpr double boltzmannDbwHzK = 228.599;
81
82 // 3. Noise Bandwidth in dBHz: 10*log10(B_Hz)
83 double bwHz = channelBandwidthKhz * 1000.0;
84 double bwDbHz = 10.0 * std::log10(bwHz);
85
86 // 4. SNR (Carrier-to-Noise Ratio C/N):
87 // SNR = EIRP - TotalLoss + G/T + 228.599 - 10*log10(B)
88 double snrDb = eirpDbw - totalLossDb + gOverTDbK + boltzmannDbwHzK - bwDbHz;
89 return snrDb;
90}

92double SignalLossModel::calculateDopplerShift(double carrierFreqMhz, double rangeRateKmS) {
93 // Doppler equation: Delta_f = - f0 * (v_los / c)
94 constexpr double cKmS = 299792.458;
95 return - (carrierFreqMhz * 1e6) * (rangeRateKmS / cKmS);
96}
